sub arithmeticOps_I_I_k_(a as Integer, b as Integer)
    sum = a + b
    diff = a - b
    product = a * b
    quotient = a / b
    remainder = a mod b
end sub

sub comparisonOps_I_I_k_(a as Integer, b as Integer)
    lt = a < b
    gt = a > b
    le = a <= b
    ge = a >= b
    eq = a = b
    ne = a <> b
end sub

sub logicalOps_Z_Z_k_(a as Boolean, b as Boolean)
    and = a and b
    or = a or b
    not = a.not()
end sub
