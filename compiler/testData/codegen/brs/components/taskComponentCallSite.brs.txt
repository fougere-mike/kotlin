            if e.message <> invalid then
                errInfo.message = e.message
            end if
            if e.number <> invalid then
                errInfo.number = e.number

                errInfo.backtrace = e.backtrace
            end if
        end if
        m.top.kotlinTaskError = errInfo
        m.top.kotlinTaskState = "error"
    end try
end sub
