function doWhileConditionScope_I_k_(seed as Integer) as Integer
    x = seed + 10
    total = x
    while true
        x_1 = total - seed
        total = (total + 1)
        if not (x_1 < 0) then
            exit while
        end if
    end while

    return total
end function
