function forLoopSum_I_k$() as Integer
    sum = 0
    for each i in 1.rangeTo(10)
        sum = (sum + i)

    end for
    return sum
end function

function whileLoopSum_I_k$() as Integer
    sum = 0
    i = 0
    while i < 10
        sum = (sum + i)
        i = (i + 1)
    end while
    return sum
end function
