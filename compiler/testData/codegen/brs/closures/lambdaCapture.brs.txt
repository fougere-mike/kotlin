function simpleCapture_k_() as Integer
    multiplier = 5
    multiply = {multiplier: multiplier, invoke: function(x as Integer) as Integer
        return x * m.multiplier
    end function}
    return multiply.invoke(10)
end function

function mutableCapture_k_() as Integer
    counter = 0
    increment = {counter: {value: counter}, invoke: function() as Void
        m.counter.value = (m.counter.value + 1)
    end function}
    increment.invoke()
    increment.invoke()
    return counter
end function

function returnedClosure_k_() as Object
    count = 0
    return {count: {value: count}, invoke: function() as Integer
        m.count.value = (m.count.value + 1)
        return m.count.value
    end function}
end function

function higherOrderFunction_I_Function1II_k_(value as Integer, transform as Object) as Integer
    return transform.invoke(value)
end function

function testHigherOrder_k_() as Integer
    return higherOrderFunction_I_Function1II_k_(5, {invoke: function(it as Integer) as Integer
        return it * 2
    end function})
end function
